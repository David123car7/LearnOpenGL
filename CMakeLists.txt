cmake_minimum_required(VERSION 3.10)
project(learn-opengl)
set(CMAKE_CXX_STANDARD_20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(GLFW3 REQUIRED)
find_package(GLAD REQUIRED)
add_executable(learn-opengl src/Main.cpp src/GLFW.cpp src/GLAD.cpp)
target_include_directories(learn-opengl PRIVATE include) 
target_link_libraries(learn-opengl PRIVATE glfw glad::glad)
